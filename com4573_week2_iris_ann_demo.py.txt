### 1. Imports

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import Perceptron
from sklearn.metrics import accuracy_score, confusion_matrix, ConfusionMatrixDisplay

"""### 2. Load a public dataset
The Iris dataset is distributed with scikit-learn, so the demo is reproducible without a separate download step.
Question: What does one row represent?

"""

iris = load_iris(as_frame=True)
df = iris.frame.copy()
df.head()

"""### 3. Inspect the data
Question: Which columns are features, and which column is the target?

"""

print("Shape:", df.shape)
print("\nFeature names:", iris.feature_names)
print("\nTarget names:", list(iris.target_names))
print("\nClass counts:")
print(df["target"].value_counts().sort_index())

df.describe()

"""### 4. Make the first problem easy to visualize
We use the first two classes and two features. This gives a binary, two-dimensional problem.
Question before plotting: Do you expect one straight line to separate the classes?

"""

subset = df[df["target"] < 2].copy()
feature_names = ["sepal length (cm)", "sepal width (cm)"]

plt.figure(figsize=(7, 5))
plt.scatter(subset[feature_names[0]], subset[feature_names[1]], c=subset["target"])
plt.xlabel(feature_names[0])
plt.ylabel(feature_names[1])
plt.title("Iris: two classes, two features")
plt.show()

"""### 5. Define X and y
X contains the inputs. y contains the answer that the model should learn to predict.

"""

X = subset[feature_names]
y = subset["target"]

print("X shape:", X.shape)
print("y shape:", y.shape)
print(X.head())
print(y.head())

"""### 6. Split into training and test sets
The test set is held out from parameter learning.

"""

X_train, X_test, y_train, y_test = train_test_split(
    X, y,
    test_size=0.25,
    random_state=42,
    stratify=y
)

print("Training samples:", len(X_train))
print("Test samples:", len(X_test))

"""### 7. Standardize the features
Fit the scaler on training data only. Then apply the same transformation to test data.

"""

scaler = StandardScaler()
X_train_s = scaler.fit_transform(X_train)
X_test_s = scaler.transform(X_test)

"""### 8. Train a perceptron
Connect this cell to the equation: prediction = step(w^T x + b).

"""

perceptron = Perceptron(max_iter=1000, tol=1e-3, random_state=42)
perceptron.fit(X_train_s, y_train)

print("Learned weights:", perceptron.coef_)
print("Learned bias:", perceptron.intercept_)

"""### 9. Evaluate unseen examples
Question: Why do we evaluate on X_test instead of X_train?

"""

y_pred = perceptron.predict(X_test_s)
acc = accuracy_score(y_test, y_pred)
cm = confusion_matrix(y_test, y_pred)

print(f"Test accuracy: {acc:.3f}")
print("Confusion matrix:\n", cm)

ConfusionMatrixDisplay(confusion_matrix=cm, display_labels=iris.target_names[:2]).plot()
plt.title("Perceptron: test confusion matrix")
plt.show()

"""### 10. Draw the learned decision boundary
For standardized features, the boundary is w1*x1 + w2*x2 + b = 0.

"""

X_all_s = scaler.transform(X)
w1, w2 = perceptron.coef_[0]
b = perceptron.intercept_[0]

x_min, x_max = X_all_s[:, 0].min() - 0.5, X_all_s[:, 0].max() + 0.5
xx = np.linspace(x_min, x_max, 200)

plt.figure(figsize=(7, 5))
plt.scatter(X_all_s[:, 0], X_all_s[:, 1], c=y)

if abs(w2) > 1e-12:
    yy = -(w1 * xx + b) / w2
    plt.plot(xx, yy, linewidth=2, label="Learned decision boundary")

plt.xlabel("Standardized sepal length")
plt.ylabel("Standardized sepal width")
plt.title("Perceptron decision boundary")
plt.legend()
plt.show()

"""### 11. Inspect predictions
Look for mistakes instead of reporting accuracy only.

"""

results = X_test.copy()
results["true_class"] = y_test.values
results["predicted_class"] = y_pred
results["correct"] = results["true_class"] == results["predicted_class"]
results

"""### 12. Optional Week 3 preview: a small ANN
Now use all four features and all three Iris classes.
This is a preview only. Hidden layers, activations, loss, and backpropagation are studied later.

"""

import tensorflow as tf

tf.keras.utils.set_random_seed(42)

X3 = df[iris.feature_names].to_numpy()
y3 = df["target"].to_numpy()

X3_train, X3_test, y3_train, y3_test = train_test_split(
    X3, y3, test_size=0.25, random_state=42, stratify=y3
)

scaler3 = StandardScaler()
X3_train_s = scaler3.fit_transform(X3_train)
X3_test_s = scaler3.transform(X3_test)

ann = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(4,)),
    tf.keras.layers.Dense(8, activation="relu"),
    tf.keras.layers.Dense(3, activation="softmax")
])

ann.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

history = ann.fit(
    X3_train_s, y3_train,
    validation_split=0.2,
    epochs=80,
    verbose=0
)

loss, ann_acc = ann.evaluate(X3_test_s, y3_test, verbose=0)
print(f"ANN test accuracy: {ann_acc:.3f}")

"""### 13. Training curves
Question: Why can training and validation performance differ?

"""

plt.figure(figsize=(7, 4))
plt.plot(history.history["accuracy"], label="Training accuracy")
plt.plot(history.history["val_accuracy"], label="Validation accuracy")
plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.title("Small ANN: training history")
plt.legend()
plt.show()
